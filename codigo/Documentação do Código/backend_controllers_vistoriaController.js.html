/**
 * @file vistoriacontroller.js
 * @namespace backend.controllers.vistoria
 * @description Controladores relacionados às operações de vistoria, como criação e agendamento por cliente.
 */

const db = require("../db");
const { enviarEmailAgendamento } = require("../enviarEmail");

/**
 * Cria uma nova vistoria pelo administrador.
 * 
 * Valida campos obrigatórios, cria o registro no banco, e envia e-mails para
 * o vistoriador e cliente informando sobre o agendamento.
 * 
 * @memberof backend.controllers.vistoria
 * @async
 * @function criarVistoria
 * @param {Object} req - Objeto de requisição HTTP (Express)
 * @param {Object} req.body
 * @param {number} req.body.idcliente - ID do cliente
 * @param {number} req.body.idvistoriador - ID do vistoriador
 * @param {number} req.body.idimovel - ID do imóvel
 * @param {string} [req.body.observacoes] - Observações adicionais
 * @param {string} req.body.datainicio - Data/hora de início da vistoria (ISO string)
 * @param {Object} res - Objeto de resposta HTTP (Express)
 * @returns {Promise<void>} Retorna resposta HTTP com o objeto da nova vistoria ou erro
 * @throws {400} Se campos obrigatórios estiverem faltando ou data inválida
 * @throws {500} Em caso de erro interno no servidor
 * @example
 * // Requisição POST /vistoria
 * {
 *   "idcliente": 1,
 *   "idvistoriador": 2,
 *   "idimovel": 3,
 *   "observacoes": "Cuidado com o piso",
 *   "datainicio": "2025-07-01T14:00:00Z"
 * }
 * @since v1.0.0
 */
async function criarVistoria(req, res) {
  const { idcliente, idvistoriador, idimovel, observacoes, datainicio } = req.body;

  if (!idcliente || !idvistoriador || !idimovel || !datainicio) {
    return res.status(400).json({ error: 'Todos os campos obrigatórios devem ser preenchidos.' });
  }

  try {
    const dataInicioTimestamp = new Date(datainicio);

    if (isNaN(dataInicioTimestamp.getTime())) {
      return res.status(400).json({ error: 'Data de início inválida.' });
    }

    const [novaVistoria] = await db`
      INSERT INTO vistoria (
        idcliente, idvistoriador, idimovel, observacoes, datainicio, status
      ) VALUES (
        ${Number(idcliente)},
        ${Number(idvistoriador)},
        ${Number(idimovel)},
        ${observacoes},
        ${dataInicioTimestamp},
        'Aguardando Agendamento da Vistoria'
      )
      RETURNING *
    `;

    const [cliente] = await db`
      SELECT nome, email FROM cliente WHERE idcliente = ${idcliente}
    `;

    const [imovel] = await db`
      SELECT descricao FROM imovel WHERE idimovel = ${idimovel}
    `;

    const [vistoriador] = await db`
      SELECT nome, email FROM funcionario WHERE id = ${idvistoriador}
    `;

    await enviarEmailAgendamento({
      para: process.env.EMAIL_USER,
      nomeCliente: cliente.nome,
      endereco: imovel.descricao,
      nomeVistoriador: vistoriador.nome,
    });

    await enviarEmailAgendamento({
      para: process.env.EMAIL_DESTINO,
      nomeCliente: cliente.nome,
      endereco: imovel.descricao,
      nomeVistoriador: vistoriador.nome,
      tipo: "cliente",
    });

    res.status(201).json(novaVistoria);
  } catch (err) {
    console.error("Erro ao agendar vistoria:", err);
    res.status(500).json({ erro: "Erro ao agendar vistoria" });
  }
}

/**
 * Permite que o cliente faça o agendamento da vistoria para um imóvel.
 * 
 * Atualiza status do imóvel para "Agendamento Pendente", envia notificação por e-mail ao vistoriador.
 * 
 * @memberof backend.controllers.vistoria
 * @async
 * @function agendarVistoriaPorCliente
 * @param {Object} req - Objeto de requisição HTTP (Express)
 * @param {Object} req.params
 * @param {number} req.params.idimovel - ID do imóvel para agendamento
 * @param {Object} req.body
 * @param {string} req.body.dataagendada - Data/hora agendada (ISO string)
 * @param {Object} res - Objeto de resposta HTTP (Express)
 * @returns {Promise<void>} Retorna mensagem de sucesso ou erro
 * @throws {500} Em caso de erro interno no servidor
 * @example
 * // Requisição PUT /vistoria/agendar/5
 * {
 *   "dataagendada": "2025-07-05T10:00:00Z"
 * }
 * @since v1.0.0
 */
async function agendarVistoriaPorCliente(req, res) {
  const { idimovel } = req.params;
  const { dataagendada } = req.body;

  try {
    await db`
      UPDATE imovel SET status = 'Agendamento Pendente', observacoes = 'Solicitado pelo cliente'
      WHERE idimovel = ${idimovel}
    `;

    const [imovel] = await db`
      SELECT descricao FROM imovel WHERE idimovel = ${idimovel}
    `;

    const [cliente] = await db`
      SELECT nome, email FROM cliente
      WHERE idcliente = (
        SELECT idcliente FROM vistoria WHERE idimovel = ${idimovel} LIMIT 1
      )
    `;

    const [vistoriador] = await db`
      SELECT nome, email FROM funcionario LIMIT 1
    `;

    await enviarEmailAgendamento({
      para: vistoriador.email,
      nomeCliente: cliente.nome,
      endereco: imovel.descricao,
      nomeVistoriador: vistoriador.nome,
    });

    res.status(200).json({ mensagem: "Solicitação enviada e e-mail notificado." });
  } catch (err) {
    console.error("Erro ao processar agendamento do cliente:", err);
    res.status(500).json({ erro: "Erro ao processar solicitação de vistoria." });
  }
}

module.exports = {

  agendarVistoriaPorCliente
