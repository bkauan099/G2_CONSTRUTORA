/**
 * @file empreendimento.js
 * @namespace backend.models.empreendimento
 * @description Rotas para operações CRUD de empreendimentos e listagem de imóveis relacionados.
 */

const express = require('express');
const router = express.Router();
const db = require('../db');
const multer = require('multer');
const path = require('path');
const fs = require('fs');

// Garante que a pasta de uploads exista
const uploadDir = path.join(__dirname, '..', 'uploads');
if (!fs.existsSync(uploadDir)) {
  fs.mkdirSync(uploadDir);
}

// Configuração do Multer
const storage = multer.diskStorage({
  destination: (req, file, cb) => cb(null, uploadDir),
  filename: (req, file, cb) => cb(null, `${Date.now()}-${file.originalname}`)
});
const upload = multer({ storage });

/**
 * Cria um novo empreendimento.
 * @memberof backend.models.empreendimento
 * @function createEmpreendimento
 * @name POST /empreendimento
 * @param {Object} req.body - Dados do empreendimento
 * @param {string} req.body.nome - Nome do empreendimento
 * @param {string} req.body.construtora - Nome da construtora
 * @param {string} req.body.estado - Estado do empreendimento
 * @param {string} req.body.cidade - Cidade do empreendimento
 * @param {string} req.body.cep - CEP do empreendimento
 * @param {string} req.body.rua - Rua do empreendimento
 * @param {File} [req.file] - Arquivo anexo (opcional)
 * @returns {Object} 201 - Empreendimento criado
 * @throws {500} Erro ao criar empreendimento
 * @example
 * // Exemplo com FormData
 * const form = new FormData();
 * form.append('nome', 'Residencial A');
 * form.append('anexos', fileInput.files[0]);
 * fetch('/empreendimento', { method: 'POST', body: form });
 * @since v1.0.0
 */
router.post('/', upload.single('anexos'), async (req, res) => {
  const { nome, construtora, estado, cidade, cep, rua } = req.body;
  const arquivoAnexo = req.file ? req.file.filename : null;

  try {
    const [empreendimento] = await db`
      INSERT INTO empreendimento
        (nome, construtora, estado, cidade, cep, rua, anexos)
      VALUES
        (${nome}, ${construtora}, ${estado}, ${cidade}, ${cep}, ${rua}, ${arquivoAnexo})
      RETURNING *
    `;
    res.status(201).json(empreendimento);
  } catch (err) {
    console.error('❌ Erro ao criar empreendimento:', err);
    res.status(500).json({ error: 'Erro ao criar empreendimento.', detalhes: err.message });
  }
});

/**
 * Lista todos os empreendimentos cadastrados.
 * @memberof backend.models.empreendimento
 * @function listEmpreendimentos
 * @name GET /empreendimento
 * @returns {Array<Object>} 200 - Lista de empreendimentos
 * @throws {500} Erro ao buscar empreendimentos
 * @example
 * fetch('/empreendimento').then(res => res.json())
 * @since v1.0.0
 */
router.get('/', async (req, res) => {
  try {
    const empreendimentos = await db`
      SELECT * FROM empreendimento ORDER BY idempreendimento DESC
    `;
    res.status(200).json(empreendimentos);
  } catch (err) {
    res.status(500).json({ error: 'Erro ao buscar empreendimentos.', detalhes: err.message });
  }
});

/**
 * Busca um empreendimento específico por ID.
 * @memberof backend.models.empreendimento
 * @function getEmpreendimentoById
 * @name GET /empreendimento/:id
 * @param {number} req.params.id - ID do empreendimento
 * @returns {Object} 200 - Dados do empreendimento
 * @throws {400} ID inválido
 * @throws {404} Empreendimento não encontrado
 * @throws {500} Erro ao buscar empreendimento
 * @example
 * fetch('/empreendimento/1').then(res => res.json())
 * @since v1.0.0
 */
router.get('/:id', async (req, res) => {
  const id = Number(req.params.id);
  if (isNaN(id)) return res.status(400).json({ error: 'ID inválido.' });

  try {
    const [empreendimento] = await db`
      SELECT * FROM empreendimento WHERE idempreendimento = ${id}
    `;
    if (!empreendimento) {
      return res.status(404).json({ error: 'Empreendimento não encontrado.' });
    }

    res.status(200).json(empreendimento);
  } catch (error) {
    res.status(500).json({ error: 'Erro ao buscar empreendimento.', detalhes: error.message });
  }
});

/**
 * Lista os imóveis vinculados a um empreendimento.
 * @memberof backend.models.empreendimento
 * @function listImoveisPorEmpreendimento
 * @name GET /empreendimento/:id/imoveis
 * @param {number} req.params.id - ID do empreendimento
 * @returns {Array<Object>} 200 - Lista de imóveis
 * @throws {400} ID inválido

router.get('/:id/imoveis', async (req, res) => {
  const id = Number(req.params.id);
  if (isNaN(id)) return res.status(400).json({ error: 'ID inválido.' });

  try {
    const imoveis = await db`
      SELECT * FROM imovel WHERE idempreendimento = ${id}
    `;
    res.status(200).json(imoveis);
  } catch (error) {
    res.status(500).json({ error: 'Erro ao buscar imóveis do empreendimento.', detalhes: error.message });
  }
});

/**
 * Atualiza os dados de um empreendimento existente.
 * @memberof backend.models.empreendimento
 * @function updateEmpreendimento
 * @name PUT /empreendimento/:id
 * @param {number} req.params.id - ID do empreendimento
 * @param {Object} req.body - Dados atualizados
 * @param {string} [req.body.nome] - Nome do empreendimento
 * @param {string} [req.body.construtora] - Nome da construtora
 * @param {string} [req.body.estado] - Estado
 * @param {string} [req.body.cidade] - Cidade
 * @param {string} [req.body.cep] - CEP
 * @param {string} [req.body.rua] - Rua
 * @param {File} [req.file] - Novo arquivo anexo (opcional)
 * @returns {Object} 200 - Mensagem de sucesso
 * @throws {400} ID inválido
 * @throws {404} Empreendimento não encontrado
 * @throws {500} Erro ao atualizar empreendimento
 * @example
 * // Enviar com FormData para atualizar com anexo
 * @since v1.0.0
 */
router.put('/:id', upload.single('anexos'), async (req, res) => {
  const id = Number(req.params.id);
  const { nome, construtora, estado, cidade, cep, rua } = req.body;
  const arquivoAnexo = req.file ? req.file.filename : null;

  if (isNaN(id)) return res.status(400).json({ error: 'ID inválido.' });

  try {
    const [empreendimentoExistente] = await db`
      SELECT * FROM empreendimento WHERE idempreendimento = ${id}
    `;
    if (!empreendimentoExistente) {
      return res.status(404).json({ error: 'Empreendimento não encontrado.' });
    }

    await db`
      UPDATE empreendimento SET
        nome = ${nome ?? empreendimentoExistente.nome},
        construtora = ${construtora ?? empreendimentoExistente.construtora},
        estado = ${estado ?? empreendimentoExistente.estado},
        cidade = ${cidade ?? empreendimentoExistente.cidade},
        cep = ${cep ?? empreendimentoExistente.cep},
        rua = ${rua ?? empreendimentoExistente.rua},
        anexos = ${arquivoAnexo ?? empreendimentoExistente.anexos}
      WHERE idempreendimento = ${id}
    `;

    res.status(200).json({ message: 'Empreendimento atualizado com sucesso.' });
  } catch (err) {
    res.status(500).json({ error: 'Erro ao atualizar empreendimento.', detalhes: err.message });
  }
});

/**
 * Remove um empreendimento existente.
 * @memberof backend.models.empreendimento
 * @function deleteEmpreendimento
 * @name DELETE /empreendimento/:id
 * @param {number} req.params.id - ID do empreendimento
 * @returns {Object} 200 - Mensagem de sucesso
 * @throws {400} ID inválido
 * @throws {404} Empreendimento não encontrado
 * @throws {500} Erro ao excluir empreendimento
 * @example
 * fetch('/empreendimento/3', { method: 'DELETE' })
 * @since v1.0.0
 */
router.delete('/:id', async (req, res) => {
  const id = Number(req.params.id);
  if (isNaN(id)) return res.status(400).json({ error: 'ID inválido.' });

  try {
    const result = await db`
      DELETE FROM empreendimento WHERE idempreendimento = ${id}
    `;
    if (result.count === 0) {
      return res.status(404).json({ error: 'Empreendimento não encontrado para exclusão.' });
    }

    res.status(200).json({ message: 'Empreendimento excluído com sucesso.' });
  } catch (err) {
    res.status(500).json({ error: 'Erro ao excluir empreendimento.', detalhes: err.message });
  }
});

module.exports = router;
