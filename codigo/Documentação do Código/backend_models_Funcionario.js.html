/**
 * @file funcionario.js
 * @namespace backend.models.funcionario
 * @description Rotas para gerenciamento de funcionários e seus respectivos cargos (administrador ou vistoriador).
 */

const express = require('express');
const router = express.Router();
const db = require('../db');

/**
 * Lista todos os funcionários com seus respectivos cargos.
 * @memberof backend.models.funcionario
 * @function listFuncionarios
 * @name GET /funcionario
 * @returns {Array<Object>} 200 - Lista de funcionários com cargo
 * @throws {500} Erro ao buscar funcionários
 * @example
 * fetch('/funcionario').then(res => res.json())
 * @since v1.0.0
 */
router.get('/', async (req, res) => {
  try {
    const funcionarios = await db`
      SELECT 
        f.*, 
        CASE 
          WHEN v.idvistoriador IS NOT NULL THEN 'Vistoriador'
          WHEN a.idadministrador IS NOT NULL THEN 'Administrador'
          ELSE 'Desconhecido'
        END AS cargo
      FROM funcionario f
      LEFT JOIN vistoriador v ON f.id = v.idvistoriador
      LEFT JOIN administrador a ON f.id = a.idadministrador
      ORDER BY f.id DESC
    `;
    res.status(200).json(funcionarios);
  } catch (err) {
    console.error('Erro ao buscar funcionários:', err);
    res.status(500).json({ error: 'Erro ao buscar funcionários.', detalhes: err.message });
  }
});

/**
 * Cadastra um novo funcionário com seu cargo.
 * @memberof backend.models.funcionario
 * @function createFuncionario
 * @name POST /funcionario
 * @param {Object} req.body - Dados do funcionário
 * @param {string} req.body.nome - Nome do funcionário
 * @param {string} req.body.cpf - CPF
 * @param {string} req.body.email - Email
 * @param {string} req.body.senha - Senha
 * @param {string} [req.body.telefone] - Telefone (opcional)
 * @param {'Administrador'|'Vistoriador'} req.body.cargo - Cargo do funcionário
 * @returns {Object} 201 - Funcionário criado com o cargo
 * @throws {400} Campos obrigatórios ausentes
 * @throws {500} Erro ao cadastrar funcionário
 * @example
 * fetch('/funcionario', {
 *   method: 'POST',
 *   headers: { 'Content-Type': 'application/json' },
 *   body: JSON.stringify({ nome, cpf, email, senha, cargo })
 * })
 * @since v1.0.0
 */
router.post('/', async (req, res) => {
  const { cpf, email, nome, senha, telefone, cargo } = req.body;

  if (!cpf || !email || !nome || !senha || !cargo) {
    return res.status(400).json({ error: 'Campos obrigatórios ausentes.' });
  }

  try {
    const [funcionario] = await db`
      INSERT INTO funcionario (cpf, email, nome, senha, telefone)
      VALUES (${cpf}, ${email}, ${nome}, ${senha}, ${telefone})
      RETURNING *
    `;

    if (cargo === 'Administrador') {
      await db`INSERT INTO administrador (idadministrador) VALUES (${funcionario.id})`;
    } else if (cargo === 'Vistoriador') {
      await db`INSERT INTO vistoriador (idvistoriador) VALUES (${funcionario.id})`;
    }

    res.status(201).json({ ...funcionario, cargo });
  } catch (err) {
    console.error('Erro ao cadastrar funcionário:', err);
    res.status(500).json({ error: 'Erro ao cadastrar funcionário.', detalhes: err.message });
  }
});

/**
 * Atualiza os dados de um funcionário existente.
 * @memberof backend.models.funcionario
 * @function updateFuncionario
 * @name PUT /funcionario/:id
 * @param {number} req.params.id - ID do funcionário
 * @param {Object} req.body - Novos dados
 * @param {string} req.body.cpf - CPF atualizado
 * @param {string} req.body.email - Email atualizado
 * @param {string} req.body.nome - Nome atualizado
 * @param {string} req.body.senha - Senha atualizada
 * @param {string} req.body.telefone - Telefone atualizado
 * @returns {Object} 200 - Funcionário atualizado
 * @throws {404} Funcionário não encontrado
 * @throws {500} Erro ao atualizar funcionário
 * @example
 * fetch('/funcionario/10', {
 *   method: 'PUT',
 *   headers: { 'Content-Type': 'application/json' },
 *   body: JSON.stringify({ nome, cpf, email, senha, telefone })
 * })
 * @since v1.0.0
 */
router.put('/:id', async (req, res) => {
  const { id } = req.params;
  const { cpf, email, nome, senha, telefone } = req.body;

  try {
    const [funcionario] = await db`
      UPDATE funcionario
      SET cpf = ${cpf}, email = ${email}, nome = ${nome}, senha = ${senha}, telefone = ${telefone}
      WHERE id = ${id}
      RETURNING *
    `;

    if (!funcionario) {
      return res.status(404).json({ error: 'Funcionário não encontrado.' });
    }

    res.json(funcionario);
  } catch (err) {
    console.error('Erro ao atualizar funcionário:', err);
    res.status(500).json({ error: 'Erro ao atualizar funcionário.', detalhes: err.message });
  }
});

/**
 * Exclui um funcionário por ID.
 * @memberof backend.models.funcionario
 * @function deleteFuncionario
 * @name DELETE /funcionario/:id
 * @param {number} req.params.id - ID do funcionário
 * @param {string} req.body.confirmacao - Deve ser 'SIM' para confirmar a exclusão
 * @returns {Object} 200 - Mensagem de sucesso
 * @throws {400} Confirmação incorreta
 * @throws {404} Funcionário não encontrado
 * @throws {500} Erro ao excluir funcionário
 * @example
 * fetch('/funcionario/5', {
 *   method: 'DELETE',
 *   headers: { 'Content-Type': 'application/json' },
 *   body: JSON.stringify({ confirmacao: 'SIM' })
 * })
 * @since v1.0.0
 */
router.delete('/:id', async (req, res) => {
  const { id } = req.params;
  const { confirmacao } = req.body;

  if (confirmacao?.trim().toUpperCase() !== 'SIM') {
    return res.status(400).json({ error: 'Confirmação incorreta. Digite "SIM" para confirmar a exclusão.' });
  }

  try {
    const result = await db`
      DELETE FROM funcionario
      WHERE id = ${id}
      RETURNING *
    `;

    if (result.length === 0) {
      return res.status(404).json({ error: 'Funcionário não encontrado.' });
    }

    res.json({ message: 'Funcionário deletado com sucesso.' });
  } catch (err) {
    console.error('Erro ao excluir funcionário:', err);
    res.status(500).json({ error: 'Erro ao excluir funcionário.', detalhes: err.message });
  }
});

/**
 * Busca um funcionário por ID, incluindo o cargo.
 * @memberof backend.models.funcionario
 * @function getFuncionarioById
 * @name GET /funcionario/:id
 * @param {number} req.params.id - ID do funcionário
 * @returns {Object} 200 - Funcionário com cargo
 * @throws {400} ID inválido
 * @throws {404} Funcionário não encontrado
 * @throws {500} Erro ao buscar funcionário
 * @example
 * fetch('/funcionario/8').then(res => res.json())
 * @since v1.0.0
 */
router.get('/:id', async (req, res) => {
  const id = Number(req.params.id);

  if (isNaN(id)) {
    return res.status(400).json({ error: 'ID inválido.' });
  }

  try {
    const [funcionario] = await db`
      SELECT 
        f.*, 
        CASE 
          WHEN v.idvistoriador IS NOT NULL THEN 'Vistoriador'
          WHEN a.idadministrador IS NOT NULL THEN 'Administrador'
          ELSE 'Desconhecido'
        END AS cargo
      FROM funcionario f
      LEFT JOIN vistoriador v ON f.id = v.idvistoriador
      LEFT JOIN administrador a ON f.id = a.idadministrador
      WHERE f.id = ${id}
    `;

    if (!funcionario) {
      return res.status(404).json({ error: 'Funcionário não encontrado.' });
    }

    res.status(200).json(funcionario);
  } catch (err) {
    console.error('Erro ao buscar funcionário por ID:', err);
    res.status(500).json({ error: 'Erro ao buscar funcionário.', detalhes: err.message });
  }
});
