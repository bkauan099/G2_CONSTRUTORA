/**
 * @file vistoria.js
 * @namespace backend.models.vistoria
 * @description Rotas para CRUD e ações específicas relacionadas a vistorias.
 */

const express = require('express');
const router = express.Router();
const db = require('../db');

/**
 * Lista todas as vistorias.
 * @memberof models.vistoria
 * @function getVistorias
 * @name GET /vistoria
 * @returns {Array<Object>} 200 - Lista de vistorias
 * @throws {500} Erro ao listar vistorias
 * @example
 * // GET /vistoria
 */
router.get('/', async (req, res) => {
  try {
    const vistorias = await db`SELECT * FROM vistoria`;
    res.status(200).json(vistorias);
  } catch (err) {
    console.error('Erro ao listar vistorias:', err);
    res.status(500).json({ error: 'Erro ao listar vistorias.' });
  }
});

/**
 * Busca uma vistoria por ID com dados completos relacionados.
 * @memberof backend.models.vistoria
 * @function getVistoriaPorId
 * @name GET /vistoria/:id
 * @param {number} req.params.id - ID da vistoria
 * @returns {Object} 200 - Vistoria encontrada
 * @throws {400} ID inválido
 * @throws {404} Vistoria não encontrada
 * @throws {500} Erro ao buscar vistoria
 * @example
 * // GET /vistoria/123
 */
router.get('/:id', async (req, res) => {
  const { id } = req.params;
  const idParsed = Number(id);

  if (isNaN(idParsed)) {
    return res.status(400).json({ error: 'ID inválido.' });
  }

  try {
    const [vistoria] = await db`
      SELECT 
        v.*,
        i.observacoes,
        v.status,
        i.descricao,
        i.bloco,
        i.numero,
        i.vistoriasrealizadas,
        e.anexos,
        e.nome AS nomeempreendimento,
        e.cidade,
        e.estado,
        e.cep,
        e.rua,
        c.nome AS nomecliente,
        c.cpf AS cpfcliente
      FROM vistoria v
      JOIN imovel i ON v.idimovel = i.idimovel
      JOIN cliente c ON i.idcliente = c.idcliente
      JOIN empreendimento e ON i.idempreendimento = e.idempreendimento
      WHERE v.idvistoria = ${idParsed}
    `;

    if (!vistoria) {
      return res.status(404).json({ error: 'Vistoria não encontrada.' });
    }

    res.status(200).json(vistoria);
  } catch (err) {
    console.error('Erro ao buscar vistoria:', err);
    res.status(500).json({ error: 'Erro ao buscar vistoria.' });
  }
});

/**
 * Agenda uma vistoria (cliente define data e hora).
 * @memberof backend.models.vistoria
 * @function agendarVistoria
 * @name PUT /vistoria/:idvistoria
 * @param {number} req.params.idvistoria - ID da vistoria
 * @param {string} req.body.dataagendada - Data agendada (YYYY-MM-DD)
 * @param {string} req.body.horaagendada - Hora agendada (HH:mm)
 * @returns {Object} 200 - Mensagem de sucesso
 * @throws {400} ID inválido ou dados obrigatórios ausentes ou inválidos
 * @throws {404} Vistoria não encontrada ou já agendada
 * @throws {500} Erro ao agendar vistoria
 * @example
 * // PUT /vistoria/123
 * // { "dataagendada": "2025-07-01", "horaagendada": "14:30" }
 */
router.put('/:idvistoria', async (req, res) => {
  const { idvistoria } = req.params;
  const { dataagendada, horaagendada } = req.body;

  const idParsed = Number(idvistoria);
  if (isNaN(idParsed)) {
    return res.status(400).json({ error: 'ID de vistoria inválido.' });
  }

  if (!dataagendada || !horaagendada) {
    return res.status(400).json({ error: 'Data e hora são obrigatórias.' });
  }

  try {
    const isoDatetimeStr = `${dataagendada}T${horaagendada}:00-03:00`;
    const dataFinal = new Date(isoDatetimeStr);

    if (isNaN(dataFinal.getTime())) {
      return res.status(400).json({ error: 'Data e hora inválidas.' });
    }

    const [vistoriaAtualizada] = await db`
      UPDATE vistoria
      SET dataagendada = ${dataFinal.toISOString()}, status = 'Vistoria Agendada'
      WHERE idvistoria = ${idParsed} AND status = 'Aguardando Agendamento da Vistoria'
      RETURNING *
    `;

    if (!vistoriaAtualizada) {
      return res.status(404).json({ error: 'Vistoria não encontrada ou já agendada.' });
    }

    res.status(200).json({ message: 'Vistoria agendada com sucesso.' });
  } catch (error) {
    console.error('Erro ao agendar vistoria:', error);
    res.status(500).json({ error: 'Erro ao agendar vistoria.' });
  }
});

/**
 * Atualiza dados de uma vistoria.
 * @memberof backend.models.vistoria
 * @function atualizarVistoria
 * @name PUT /vistoria/:id
 * @param {number} req.params.id - ID da vistoria
 * @param {Object} req.body - Dados da vistoria para atualizar
 * @param {string} [req.body.dataagendada]
 * @param {string} [req.body.datahorainicio]
 * @param {string} [req.body.datahorafim]
 * @param {string} [req.body.condicoesclimaticas]
 * @param {string} [req.body.imprevistos]
 * @param {string} [req.body.observacoesgerais]
 * @param {string} [req.body.status]
 * @param {string} [req.body.relatorio_url]
 * @returns {Object} 200 - Mensagem de sucesso
 * @throws {400} ID inválido
 * @throws {404} Vistoria não encontrada
 * @throws {500} Erro ao atualizar vistoria
 * @example
 * // PUT /vistoria/123
 * // { "status": "Finalizada" }
 */
router.put('/:id', async (req, res) => {
  const { id } = req.params;
  const idParsed = Number(id);
  const {
    dataagendada,
    datahorainicio,
    datahorafim,
    condicoesclimaticas,
    imprevistos,
    observacoesgerais,
    status,
    relatorio_url
  } = req.body;

  if (isNaN(idParsed)) {
    return res.status(400).json({ error: 'ID inválido.' });
  }

  try {
    const [vistoriaExistente] = await db`SELECT * FROM vistoria WHERE idvistoria = ${idParsed}`;

    if (!vistoriaExistente) {
      return res.status(404).json({ error: 'Vistoria não encontrada.' });
    }

    await db`
      UPDATE vistoria SET
        dataagendada = ${dataagendada ?? vistoriaExistente.dataagendada},
        datahorainicio = ${datahorainicio ?? vistoriaExistente.datahorainicio},
        datahorafim = ${datahorafim ?? vistoriaExistente.datahorafim},
        condicoesclimaticas = ${condicoesclimaticas ?? vistoriaExistente.condicoesclimaticas},
        imprevistos = ${imprevistos ?? vistoriaExistente.imprevistos},
        observacoesgerais = ${observacoesgerais ?? vistoriaExistente.observacoesgerais},
        status = ${status ?? vistoriaExistente.status},
        relatorio_url = ${relatorio_url ?? vistoriaExistente.relatorio_url}
      WHERE idvistoria = ${idParsed}
    `;

    res.status(200).json({ message: 'Vistoria atualizada com sucesso.' });
  } catch (err) {
    console.error('Erro ao atualizar vistoria:', err);
    res.status(500).json({ error: 'Erro ao atualizar vistoria.' });
  }
});

/**
 * Valida uma vistoria, alterando status para "Finalizada" e salvando data de término.
 * @memberof backend.models.vistoria
 * @function validarVistoria
 * @name PUT /vistoria/validar/:idvistoria
 * @param {number} req.params.idvistoria - ID da vistoria
 * @returns {Object} 200 - Mensagem de sucesso
 * @throws {400} ID inválido
 * @throws {404} Vistoria não encontrada
 * @throws {500} Erro ao validar vistoria
 * @example
 * // PUT /vistoria/validar/123
 */
router.put('/validar/:idvistoria', async (req, res) => {
  const { idvistoria } = req.params;
  const idParsed = Number(idvistoria);

  if (isNaN(idParsed)) {
    return res.status(400).json({ error: 'ID de vistoria inválido.' });
  }

  try {
    const [vistoria] = await db`SELECT idimovel FROM vistoria WHERE idvistoria = ${idParsed}`;

    if (!vistoria) {
      return res.status(404).json({ error: 'Vistoria não encontrada.' });
    }

    const now = new Date().toISOString();

    await db`
      UPDATE vistoria
      SET status = 'Finalizada',
          datahorafim = ${now}
      WHERE idvistoria = ${idParsed}
    `;

    res.status(200).json({ mensagem: 'Vistoria validada com sucesso.' });
  } catch (error) {
    console.error('Erro ao validar vistoria:', error);
    res.status(500).json({ error: 'Erro ao validar vistoria.' });
  }
});
