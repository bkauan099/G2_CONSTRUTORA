/**
 * @file ReagendamentoVistoria.jsx
 * @namespace src.pages.homevistoriador.reagendarvistoria
 * @description Página para reagendamento de uma vistoria existente, permitindo alterar data, hora e motivo do reagendamento.
 */

import React, { useState, useEffect } from 'react';
import { useNavigate, useParams } from 'react-router-dom';
import './ReagendarVistoriaPage.css'; // CSS específico para esta página

/**
 * Componente que exibe formulário para reagendar uma vistoria com base no ID da URL.
 *
 * @function ReagendamentoVistoria
 * @memberof src.pages.homevistoriador.reagendarvistoria
 * @param {Object} props - Props do componente.
 * @param {function} props.onLogout - Função para deslogar o usuário.
 * @returns {JSX.Element} Formulário de reagendamento de vistoria.
 */
function ReagendamentoVistoria({ onLogout }) {
  const navigate = useNavigate();

  /**
   * ID da vistoria obtido da URL.
   * @type {string}
   */
  const { vistoriaId } = useParams();

  /**
   * Estado com dados do formulário e detalhes da vistoria.
   * @typedef {Object} SurveyDetails
   * @property {string} id - ID da vistoria.
   * @property {string} currentDate - Data atual agendada da vistoria.
   * @property {string} currentTime - Hora atual agendada da vistoria.
   * @property {string} newDate - Nova data para reagendamento.
   * @property {string} newTime - Nova hora para reagendamento.
   * @property {string} reason - Motivo do reagendamento.
   * @property {string} comments - Comentários adicionais.
   */
  const [surveyDetails, setSurveyDetails] = useState({
    id: vistoriaId,
    currentDate: '',
    currentTime: '',
    newDate: '',
    newTime: '',
    reason: '',
    comments: ''
  });

  /**
   * Estado para controle de loading.
   * @type {boolean}
   */
  const [loading, setLoading] = useState(true);

  /**
   * Estado para mensagem de erro.
   * @type {string}
   */
  const [error, setError] = useState('');

  /**
   * Estado para mensagem de sucesso.
   * @type {string}
   */
  const [successMessage, setSuccessMessage] = useState('');

  /**
   * Efeito para simular fetch dos dados atuais da vistoria com base no ID.
   * Atualiza os estados de surveyDetails, loading, error.
   */
  useEffect(() => {
    const fetchSurveyDetails = async () => {
      setLoading(true);
      setError('');

      // Simulação de fetch - substituir por chamada real à API conforme necessário
      setTimeout(() => {
        if (vistoriaId) {
          setSurveyDetails(prev => ({
            ...prev,
            currentDate: '2025-07-10', // exemplo
            currentTime: '14:30',      // exemplo
          }));
        } else {
          setError('ID da vistoria não fornecido na URL.');
        }
        setLoading(false);
      }, 1000);
    };

    fetchSurveyDetails();
  }, [vistoriaId]);

  /**
   * Manipula mudanças nos campos do formulário, atualizando o estado surveyDetails.
   *
   * @param {React.ChangeEvent<HTMLInputElement|HTMLSelectElement|HTMLTextAreaElement>} e Evento de mudança do input
   */
  const handleChange = (e) => {
    const { name, value } = e.target;
    setSurveyDetails(prev => ({
      ...prev,
      [name]: value
    }));
  };

  /**
   * Manipula o envio do formulário, validando dados e simulando a chamada para API para reagendar vistoria.
   *
   * @param {React.FormEvent<HTMLFormElement>} e Evento do formulário
   */
  const handleSubmit = async (e) => {
    e.preventDefault();
    setError('');
    setSuccessMessage('');

    // Validação básica
    if (!surveyDetails.newDate || !surveyDetails.newTime || !surveyDetails.reason) {
      setError('Por favor, preencha a nova data, hora e o motivo do reagendamento.');
      return;
    }

    setLoading(true);

    // Simulação de chamada API
    console.log("Tentando reagendar vistoria:", surveyDetails);
    setTimeout(() => {
      setLoading(false);
      setSuccessMessage(`Vistoria reagendada com sucesso para ${surveyDetails.newDate} às ${surveyDetails.newTime}.`);
    }, 1500);
  };

  if (loading && !surveyDetails.currentDate) {
    return <div className="loading-container">Carregando detalhes da vistoria...</div>;
  }

  return (
    <div className="reschedule-page-container">
      <header className="navbar">
        <div className="logo">CIVIS</div>
        <nav className="nav-links">
          <a href="#" onClick={() => navigate("/home")}>Home</a>
          <a href="#" onClick={() => navigate("/vistoriador/realizar-vistoria")}>Realizar Vistoria</a>
        </nav>
        <button className="logout-button" onClick={onLogout}>Sair</button>
      </header>

      <main className="reschedule-main-content">
        <div className="reschedule-form-card">
          <h1>Reagendamento de Vistoria {vistoriaId ? `(ID: ${vistoriaId})` : ''}</h1>
          <p className="current-details">
            Data e Hora Atuais: <strong>{surveyDetails.currentDate || 'N/A'}</strong> às <strong>{surveyDetails.currentTime || 'N/A'}</strong>
          </p>

          {error && <p className="error-message">{error}</p>}
          {successMessage && <p className="success-message">{successMessage}</p>}

          <form onSubmit={handleSubmit}>
            <div className="form-group">
              <label htmlFor="newDate">Nova Data:</label>
              <input
                type="date"
                id="newDate"
                name="newDate"
                value={surveyDetails.newDate}
                onChange={handleChange}
                required
              />
            </div>

            <div className="form-group">
              <label htmlFor="newTime">Nova Hora:</label>
              <input
                type="time"
                id="newTime"
                name="newTime"
                value={surveyDetails.newTime}
                onChange={handleChange}
                required
              />
            </div>

            <div className="form-group">
              <label htmlFor="reason">Motivo do Reagendamento:</label>
              <select
                id="reason"
                name="reason"
                value={surveyDetails.reason}
                onChange={handleChange}
                required
              >
                <option value="">Selecione um motivo</option>
                <option value="Indisponibilidade do vistoriador">Indisponibilidade do vistoriador</option>
                <option value="Indisponibilidade do cliente">Indisponibilidade do cliente</option>
                <option value="Problemas climáticos">Problemas climáticos</option>
                <option value="Outros">Outros</option>
              </select>
            </div>

            <div className="form-group">
              <label htmlFor="comments">Comentários Adicionais (Opcional):</label>
              <textarea
                id="comments"
                name="comments"
                value={surveyDetails.comments}
                onChange={handleChange}
                rows="4"
              />
            </div>

            <div className="form-actions">
              <button type="submit" className="submit-button" disabled={loading}>
                {loading ? 'Reagendando...' : 'Reagendar Vistoria'}
              </button>
              <button type="button" className="cancel-button" onClick={() => navigate(-1)}>
                Cancelar
              </button>
            </div>
          </form>
        </div>
      </main>
    </div>
  );
}

export default ReagendamentoVistoria;
