/**
 * @file Login.jsx
 * @namespace src.pages.login
 * @description Página de autenticação da aplicação CIVIS. Realiza login para cliente, funcionário ou administrador.
 */

import { useNavigate } from "react-router-dom";
import { useState } from "react";
import "./Login.css"; // Estilização com grid responsivo

/**
 * Componente de login da aplicação.
 *
 * Permite que diferentes tipos de usuários (cliente, funcionário, admin) façam login.
 * Redireciona para a home correspondente após autenticação.
 *
 * @component
 * @memberof src.pages.login
 * @param {Object} props
 * @param {function} props.onLogin - Função de callback acionada ao fazer login com sucesso.
 * @returns {JSX.Element} Componente de tela de login.
 */
function Login({ onLogin }) {
  const navigate = useNavigate();

  /**
   * Estado que controla o carregamento (spinner) durante a requisição de login.
   * @type {boolean}
   */
  const [loading, setLoading] = useState(false);

  /**
   * Lida com o envio do formulário de login.
   * Realiza uma requisição POST para autenticar o usuário.
   *
   * @async
   * @function handleLogin
   * @memberof src.pages.login
   * @param {React.FormEvent<HTMLFormElement>} e Evento do formulário.
   */
  const handleLogin = async (e) => {
    e.preventDefault();

    // Extrai os campos diretamente pelos elementos (alternativa ao useRef ou useState individual)
    const email = e.target.elements[0].value;
    const senha = e.target.elements[1].value;

    setLoading(true); // Exibe "Entrando..."

    try {
      const response = await fetch("http://localhost:3001/api/login", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ email, senha }),
      });

      const data = await response.json();

      if (!response.ok) {
        throw new Error(data.erro || "Erro desconhecido.");
      }

      const tipo = data.tipo;

      // Armazena ID se for cliente
      if (tipo === "cliente") {
        localStorage.setItem("idcliente", data.id);
      }

      alert(`Login de ${tipo} realizado com sucesso!`);
      onLogin(tipo); // Callback do componente pai
      navigate("/home"); // Redireciona para a home padrão
    } catch (err) {
      alert(err.message);
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="login-page">
      {/* Seção lateral da logo */}
      <div className="login-logo-section">
        <img
          src="assets/imagensLogin/logo.png"
          alt="CIVIS Logo"
          className="logo-image"
        />
      </div>

      {/* Container do formulário */}
      <div className="login-content-wrapper">
        <div className="login-container">
          {/* Botão de voltar para a tela inicial */}
          <button
            type="button"
            className="back-arrow"
            onClick={() => navigate("/")}
            aria-label="Voltar"
          >
            &#8592;
          </button>

          <h1 className="login-title">Login</h1>

          {/* Formulário de login */}
          <form onSubmit={handleLogin} className="login-form">
            <label>Email</label>
            <input type="email" placeholder="Digite seu email" required />

            <label>Senha</label>
            <input type="password" placeholder="Digite sua senha" required />

            {/* Botão de envio */}
            <button type="submit" className="login-button">
              {loading ? "Entrando..." : "Entrar"}
            </button>

            {/* Link para página de cadastro */}
            <p
              className="no-account"
              onClick={() => navigate("/cadastro-login")}
            >
              Não possui cadastro?
            </p>
          </form>
        </div>
      </div>
    </div>
  );
}

export default Login;
